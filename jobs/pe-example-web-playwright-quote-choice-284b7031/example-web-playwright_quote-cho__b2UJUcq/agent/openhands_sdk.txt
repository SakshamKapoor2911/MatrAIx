    def fetch_all_quotes():
        quotes = []
        with sync_playwright() as p:
            browser = p.chromium.launch(headless=True)
            page = browser.new_page()
    
            page_num = 1
            while True:
                url = f"https://quotes.toscrape.com/page/{page_num}/"
                page.goto(url, wait_until="domcontentloaded")
    
                quote_divs = page.query_selector_all("div.quote")
                if not quote_divs:
                    break
    
                for div in quote_divs:
                    text_el = div.query_selector("span.text")
                    author_el = div.query_selector("small.author")
                    tag_els = div.query_selector_all("a.tag")
    
                    text = text_el.inner_text() if text_el else ""
                    author = author_el.inner_text() if author_el else ""
                    tags = [t.inner_text() for t in tag_els]

                        "author": author,

                        "page": page_num

                # Check for next page

                if not next_btn:
